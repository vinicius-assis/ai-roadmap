Consumindo um corpo HTTP streamado

O encanamento comum a Ollama (NDJSON) e às APIs SSE (Anthropic, OpenAI).

O padrão em 4 passos

PassoCódigo
1. Abrir o streamconst res = await fetch(url, { ..., signal: AbortSignal.timeout(30_000) })
2. Iterar os bytesfor await (const bytes of res.body as AsyncIterable<Uint8Array>)
3. Decodificar + acumularbuffer += decoder.decode(bytes, { stream: true })
4. Cortar na fronteiraconst partes = buffer.split(SEP); buffer = partes.pop() ?? "";

Esqueleto reutilizável

async function* streamLines(
  url: string,
  body: unknown,
  sep = "\n",
): AsyncGenerator<string> {
  const res = await fetch(url, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(body),
    signal: AbortSignal.timeout(30_000),
  });
  if (!res.ok) throw new Error(`HTTP ${res.status}`);

  const decoder = new TextDecoder();
  let buffer = "";
  for await (const bytes of res.body as AsyncIterable<Uint8Array>) {
    buffer += decoder.decode(bytes, { stream: true });
    const partes = buffer.split(sep);
    buffer = partes.pop() ?? "";          // fatia incompleta volta pro buffer
    for (const parte of partes) {
      if (parte.trim()) yield parte;
    }
  }
  if (buffer.trim()) yield buffer;         // resto após o último separador
}
